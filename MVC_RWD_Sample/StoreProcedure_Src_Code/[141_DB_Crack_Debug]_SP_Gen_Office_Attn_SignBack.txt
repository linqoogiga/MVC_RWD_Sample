					    if(CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),120)=@Btw_Leave_ToDate)

							set @New_Hour_Unit_Leave_FromTime ='10:00'; 
							set @New_Hour_Unit_Leave_ToTime ='19:00';		
						end
					end
					else if(@Btw_Leave_FromTime='PM' and @Btw_Leave_ToTime='PM')
					begin
						--如果請假:起始日期yyyy-MM-dd PM ~ 結束日期yyyy-MM-dd PM:
					    if(CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),120)=@Btw_Leave_FromDate)
						begin
						    --如果今天是假期的第1天:
							set @New_Hour_Unit_Leave_FromTime ='15:00'; 
							set @New_Hour_Unit_Leave_ToTime ='19:00';						
						end
						else
						begin
						    --如果今天是假期的最後1天_或是_位於[請假區間]中間的日期:
							set @New_Hour_Unit_Leave_FromTime ='10:00'; 
							set @New_Hour_Unit_Leave_ToTime ='19:00';		
						end
					end
					else if(@Btw_Leave_FromTime='PM' and @Btw_Leave_ToTime='AM')
					begin
						--如果請假:起始日期yyyy-MM-dd PM ~ 結束日期yyyy-MM-dd PM:
					    if(CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),120)=@Btw_Leave_FromDate)
