DECLARE MyCursor CURSOR FOR 

SELECT AA.za40_usid  EmpId,
	   AA.za40_name  EmpName,
	   AA.za40_email Email	  
FROM 
(
	SELECT * ,
		(SELECT sa01_clas FROM blspostwn.dbo.tb_sa01(NOLOCK) 
		  WHERE sa01_faid=za40_usid) EmpType 
	  FROM blspostwn.dbo.tb_za40(NOLOCK) 
     WHERE za40_slyn='Y' 
       AND za40_sosk='1'
       AND za40_usid LIKE '2298%'
	   --AND za40_dpcd<>'205' --倉務部 20250923 Michelle 加入倉庫
	   --AND za40_usid NOT IN ('2298311123','2298310051','2298311090') --林謀嘉 / 張嘉珍 /Tammy Yu   
	   --AND za40_usid NOT IN ('2298311123','2298311090') --林謀嘉 / Tammy Yu   
                     AND za40_usid NOT IN ('2298110305','2298110306') --林謀嘉 / Tammy Yu   2022.10.13 發現換工號
) AA 
WHERE AA.EmpType='O'  
  AND AA.za40_grad<>'G1'
  --AND AA.za40_usid='2298311094'

ORDER BY AA.za40_grad


Open MyCursor 
DECLARE @ID NVARCHAR(50)
DECLARE @NA NVARCHAR(50)
DECLARE @EM NVARCHAR(50)

Fetch NEXT FROM MyCursor INTO @ID,@NA,@EM
While (@@FETCH_STATUS <> -1)
BEGIN	
	PRINT @NA
	DECLARE @INTTY INT
	INSERT INTO dbo.Attn_HDR
	(
	    UserNo,
	    UserName,
		Email,
	    PeriodName,
	    SignFlag,
	    SignDate,
	    SignBy,
	    Created,
	    Creator,
	    LastModified,
	    LastModifiedBy,
	    StatusId,
		MailFlag
	)
	VALUES
	(   @ID,        -- UserNo - nvarchar(10)
	    @NA,        -- UserName - nvarchar(20)
		@EM,	    
		@Period,
	    N'N',       -- SignFlag - nvarchar(10)
	    NULL,       -- SignDate - datetime
	    N'',        -- SignBy - nvarchar(10)
	    GETDATE(),  -- Created - datetime
	    N'System',  -- Creator - nvarchar(20)
	    GETDATE(),  -- LastModified - datetime
	    N'System',  -- LastModifiedBy - nvarchar(20)
	    1        ,   -- StatusId - int
		'N'
	    )
	SELECT @INTTY = @@IDENTITY 
	--PRINT @INTTY
	DECLARE TCursor CURSOR FOR 
	SELECT * FROM @D
	Open TCursor 
	DECLARE @DD INT
	Fetch NEXT FROM TCursor INTO @DD
	WHILE (@@FETCH_STATUS <> -1)
		BEGIN
			PRINT CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),112)
			DECLARE @IN  NVARCHAR(5)
			DECLARE @OUT NVARCHAR(5)
			DECLARE @WT  DECIMAL(8,2)
			DECLARE @HD  DECIMAL(8,2)
			DECLARE @SU  DECIMAL(8,2)
			SELECT @IN = ISNULL(CONVERT(NVARCHAR(20),MIN(DoorTime),108),''),
			       @OUT= ISNULL(CONVERT(NVARCHAR(20),MAX(DoorTime),108),''),
				   @WT = DATEDIFF(MINUTE,MIN(DoorTime),MAX(DoorTime))/60.00 ,
				   @HD = (SELECT da01_type FROM blspostwn.dbo.rp_da01(NOLOCK) WHERE da01_date=CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),112)),
				   @SU = (SELECT COUNT(*) FROM dbo.AttnRemark(NOLOCK) WHERE WorkDate>=@WD AND EmpNo=@ID)
			  FROM BLSOmnibackend.dbo.AttnLog(NOLOCK) 
			 WHERE EmpNo=@ID 
			   AND WorkDate=CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),112)     
		--IF @HD = '01' 
		--BEGIN									
			--請假
			DECLARE @RK NVARCHAR(200)   		
			SET @RK = ''	
			SELECT @RK=ISNULL(BB.myLeave,'') FROM (
			SELECT (	
			SELECT (AA.LeaveCode +'-'+ CASE WHEN AA.FromTime=AA.ToTime THEN AA.FromTime ELSE AA.FromTime+'/'+AA.ToTime END+' ')
			FROM (
			SELECT LeaveCode  AS LeaveCode,
			       FromDate,ToDate,
				   FromTime,ToTime
			  FROM dbo.OfficeApprovedLeaveView(NOLOCK) WHERE UserNo=@ID
			  AND CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),120) BETWEEN FromDate AND ToDate			
			  AND (RelatedFlowCaseId IS NULL OR RelatedFlowCaseId = 0)
			  AND FlowCaseId NOT IN (SELECT ISNULL(RelatedFlowCaseId,0) FROM dbo.OfficeApprovedLeaveView(NOLOCK) WHERE UserNo=@ID)			  
			) AA ORDER BY FromTime FOR XML PATH('') ) AS myLeave
			) BB
			--主任排班
			DECLARE @SF NVARCHAR(100)
			SET @SF = ''
			SELECT @SF = Remark
			  FROM AttnRemark(NOLOCK)
			 WHERE WorkDate = CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),112)
			   AND EmpNo = @ID
			 IF @SF<>''
				SET @RK = @RK + @SF
			--補刷卡
			DECLARE @ClockIn NVARCHAR(30)
			SET @ClockIn =''
			SELECT @ClockIn =AA.ClockInTime FROM (
			SELECT   
				--CASE WHEN b.OnTime IS NULL AND b.OffTime IS NOT NULL THEN CONVERT(NVARCHAR(5),b.OffTime)
				--	 WHEN b.OnTime IS NOT NULL AND b.OffTime IS NULL THEN CONVERT(NVARCHAR(5),b.OnTime)
			 --        ELSE CONCAT(CONVERT(NVARCHAR(5),b.OnTime),'/',CONVERT(NVARCHAR(5),b.OffTime))  
			 --   END AS ClockInTime       
			CASE WHEN MIN(b.OnTime) IS NULL AND MAX(b.OffTime) IS NOT NULL THEN CONVERT(NVARCHAR(5),MAX(b.OffTime))
					 WHEN MIN(b.OnTime) IS NOT NULL AND MAX(b.OffTime) IS NULL THEN CONVERT(NVARCHAR(5),MIN(b.OnTime))
			         ELSE CONCAT(CONVERT(NVARCHAR(5),MIN(b.OnTime)),'/',CONVERT(NVARCHAR(5),MAX(b.OffTime)))  
			END AS ClockInTime       
            FROM BLSOmnibackend.dbo.Complement_ClockIn_Hdr(NOLOCK) a ,
                 BLSOmnibackend.dbo.Complement_ClockIn_Detl(NOLOCK) b,
	             BLSOmnibackend.dbo.WF_FlowCases(NOLOCK) c
           WHERE a.HdrId = B.HdrId
             AND a.FlowCaseId = c.FlowCaseId
             AND C.Approved IS NOT NULL
             AND a.Username=@ID
			 AND Date = CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),120)
			 ) AA
			 IF @ClockIn<>'/'
				SET @RK = @RK + N' 補打卡:'+@ClockIn
			--加班
			DECLARE @OverTime NVARCHAR(30)
			SET @OverTime = ''
			--加班單顯示範例 : 加班: 19:00 – 22:00 
			SELECT @OverTime = CONCAT(AA.FromTime,'-',AA.ToTime)
			FROM (   
					SELECT 
							CONVERT(NVARCHAR(10),(SELECT B.DateValue 
							                        FROM BLSOmnibackend.dbo.WF_FlowPropertys(NOLOCK) A , 
													     BLSOmnibackend.dbo.WF_CasePropertyValues(NOLOCK) B
			                                       WHERE a.FlowPropertyId = b.PropertyId
			                                         AND PropertyName='Date'
			                                         AND B.FlowCaseId = C.FlowCaseId
		                     ) ,120) AS OTDate,
							 (SELECT StringValue 
							    FROM BLSOmnibackend.dbo.WF_FlowPropertys(NOLOCK) A , 
		                             BLSOmnibackend.dbo.WF_CasePropertyValues(NOLOCK) B
		                       WHERE a.FlowPropertyId = b.PropertyId
			                     AND PropertyName='From'
			                     AND B.FlowCaseId = C.FlowCaseId
							 ) AS FromTime,
							 (SELECT StringValue 
		                        FROM BLSOmnibackend.dbo.WF_FlowPropertys(NOLOCK) A , 
		                             BLSOmnibackend.dbo.WF_CasePropertyValues(NOLOCK) B
		                       WHERE a.FlowPropertyId = b.PropertyId
			                     AND PropertyName='To'
			                     AND B.FlowCaseId = C.FlowCaseId
		                      ) AS ToTime								
					FROM BLSOmnibackend.dbo.WF_FlowCases(NOLOCK) C
				   WHERE C.UserNo = @ID
				     AND C.StatusId>0
                     AND Created>='2022-01-01'
					 AND C.Approved IS NOT NULL
                     AND C.FlowId IN (
                         SELECT FlowId 
						   FROM BLSOmnibackend.dbo.WF_Flows(NOLOCK) 
                          WHERE FlowGroupId IN (
							SELECT FlowGroupId
							  FROM BLSOmnibackend.dbo.WF_FlowGroups(NOLOCK) 
                             WHERE FlowTypeId IN (
                              SELECT FlowTypeId 
                                FROM BLSOmnibackend.dbo.WF_FlowTypes(NOLOCK) 
                               WHERE Name=N'Overtime' 
                                 AND StatusId=1 
                            )))) AA
	         WHERE AA.OTDate = CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),120)

			 IF @OverTime<>''
				SET @RK = @RK + N' 加班:'+@OverTime
			 --假日也要有資料
			 PRINT @HD
			 PRINT @SU
			 IF (@HD<>'01' AND @SU<=0) SET @RK=N'假日'+@RK
			 IF @RK ='' AND (@IN='' OR @OUT='' OR (@IN=@OUT)) SET @RK=N'異常'

		INSERT INTO dbo.Attn_DETL
		(
		    HDR_Id,
		    WorkDate,
		    InTime,
		    OutTime,
		    WorkHours,
		    Reason,
		    Remark,
		    SysRemark,
		    Created,
		    Creator,
		    LastModified,
		    LastModifiedBy,
		    StatusId
		)
		VALUES
		(   @INTTY,         -- HDR_Id - int
		    CONVERT(NVARCHAR(10),DATEADD(DAY,@DD,@STD),112),       -- WorkDate - nvarchar(10)
		    @IN,      -- InTime - nvarchar(10)
		    @OUT,     -- OutTime - nvarchar(10)
		    @WT,      -- WorkHours - decimal(8, 2)
		    N'',       -- Reason - nvarchar(30)
		    --N'',       -- Remark - nvarchar(30)
			ISNULL(@RK,''),
		    CASE WHEN @WT>9.5 THEN 'OT' ELSE '' END,       -- SysRemark - nvarchar(30)
		    GETDATE(), -- Created - datetime
		    N'SYSTEM',       -- Creator - nvarchar(20)
		    GETDATE(), -- LastModified - datetime
		    N'SYSTEM',       -- LastModifiedBy - nvarchar(20)
		    1          -- StatusId - int
		)
		--END
		Fetch NEXT FROM TCursor INTO @DD
		END
		CLOSE TCursor
		DEALLOCATE TCursor
