ALTER FUNCTION [dbo].[fn_Get_Leave_Info]
(
    @UserNo NVARCHAR(100),
    @FromDate DATETIME
)
RETURNS TABLE
AS
RETURN
(
    SELECT DISTINCT *
    FROM
    (
        SELECT
            wffc.FlowCaseId,
            wfcnr.Receiver,
            wffc.RelatedFlowCaseId,

            (
                SELECT DateValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'FromDate'
            ) AS FromDate,

            (
                SELECT StringValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'FromTime'
            ) AS FromTime,

            (
                SELECT DateValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'ToDate'
            ) AS ToDate,

            (
                SELECT StringValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'ToTime'
            ) AS ToTime,

            (
                SELECT StringValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'LeaveCode'
            ) AS LeaveCode,

            (
                SELECT StringValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'IsMenstrualLeave'
            ) AS IsMenstrualLeave,

            (
                SELECT NumericValue
                FROM WF_CasePropertyValues wfcpv
                LEFT JOIN WF_FlowPropertys wffp
                    ON wfcpv.PropertyId = wffp.FlowPropertyId
                WHERE wfcpv.FlowCaseId = wffc.FlowCaseId
                  AND wffp.PropertyName = 'TotalHours'
            ) AS TotalHours

        FROM WF_CaseNotifications wfcn
        LEFT JOIN WF_CaseNotificationReceivers wfcnr
            ON wfcn.CaseNotificationId = wfcnr.CaseNotificationId
        LEFT JOIN WF_FlowCases wffc
            ON wfcn.FlowCaseId = wffc.FlowCaseId
        LEFT JOIN WF_CasePropertyValues wfcpv
            ON wfcpv.FlowCaseId = wfcn.FlowCaseId
        LEFT JOIN WF_FlowPropertys wffp
            ON wfcpv.PropertyId = wffp.FlowPropertyId

        WHERE
            wfcnr.IsDissmissed IN (0,1)
            AND wfcnr.StatusId = 1
            AND wfcn.StatusId > 0
            AND wffc.UserNo = @UserNo
            AND wfcnr.Receiver = @UserNo
            AND wfcpv.StringValue IS NOT NULL
            AND wfcpv.StringValue <> ''
            AND Approved IS NOT NULL
            AND Approved <> ''
            AND NotificationType = '6'
            AND wffc.StatusId > 0
    ) A
    WHERE
        A.FromDate >= @FromDate
        AND A.FromDate < DATEADD(DAY, 1, @FromDate)        
		and (A.RelatedFlowCaseId is null or A.RelatedFlowCaseId = 0)  --排除本身[WF_FlowCases]是代表Cancel的假期。
       and A.FlowCaseId NOT IN (
    --排除已經Cancel的假期。
	SELECT ISNULL(RelatedFlowCaseId, 0)
	FROM dbo.OfficeApprovedLeaveView(NOLOCK)
	WHERE UserNo=@UserNo
)
);
GO
